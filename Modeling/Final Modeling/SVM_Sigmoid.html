data <- data[, 1:45]
data$target <- as.factor(data$target)

normalizer <- function(x) {
  return((x-mean(x))/sd(x))

Modeling PseudoCode

n = nrow(data)
K = 10 # 10-fold CV

# 설정하려는 Seed 총 개수
Seed_Num = 10
  
# Seed, parameter 별로 일단 모두 저장
Accuracy_SVMS <- vector(mode = "list", length = Seed_Num)
FPR_SVMS <- vector(mode = "list", length = Seed_Num)
FNR_SVMS <- vector(mode = "list", length = Seed_Num)
F1_SVMS <- vector(mode = "list", length = Seed_Num)

# Parameter grid 설정
cost_grid <- c(0.001,0.01,0.1,1,4,16)
gamma_grid <- c(0.0001, 0.001, 0.01, 0.1)

for(seed in 1:Seed_Num){  # Seed를 다르게 하는 For loop
  print("seed")
  print(seed)
  set.seed(seed)
  
  # Parameter 별로 For loop
  for(cost in cost_grid){
    for(gamma in gamma_grid){
      
      print("cost")
      print(cost)
      print("gamma")
      print(gamma)
      
      for(CV in 1:K){
        ########################################################################################
        # CV generation. Train : Test를 9:1로 나눔 (이때의 train, test는 validation approach)  #
        ########################################################################################
        print("CV")
        print(CV)
        
        CV_index <- split(sample(1:n), rep(1:K, length.out = n))
        train <- data[-CV_index[[CV]], ] # Train set
        train_index <- c(1:n) 
        train_index <- train_index[-CV_index[[CV]]] # Train set의 index (필요할 경우 사용)
        test <- data[CV_index[[CV]], ] # Test set
        test_rowNum <- nrow(test) # Test set row Numbers (필요할 경우 사용))
        
        #################################################################
        # Modeling and Prediction. Update Accuracy, FNR, FPR, F1 score  #
        #################################################################
        # Train 데이터셋이 필요한 경우
        Model <- svm(target ~ ., data = train, cost = cost, gamma = gamma, kernel = "sigmoid")
        
        
        # Prediction (0, 1로 나오는 경우)
        pred <- predict(Model, newdata = test)
        
        # Update Scores
        Accuracy_SVMS[[seed]] <- c(Accuracy_SVMS[[seed]], mean(pred == test$target))
        FPR_SVMS[[seed]] <- c(FPR_SVMS[[seed]], mean((pred != test$target) & (test$target == 0)) / mean(test$target == 0))
        FNR_SVMS[[seed]] <- c(FNR_SVMS[[seed]], mean((pred != test$target) & (test$target == 1)) / mean(test$target == 1))
        F1_SVMS[[seed]] <- c(F1_SVMS[[seed]], 2*(mean((pred==test$target)&(test$target==1))/(mean(pred==1))*mean((pred==test$target)&(test$target==1))/(mean(test$target==1))) / (mean((pred==test$target)&(test$target==1))/(mean(pred==1))+mean((pred==test$target)&(test$target==1))/(mean(test$target==1))))
      }
      
    }
  }
}
