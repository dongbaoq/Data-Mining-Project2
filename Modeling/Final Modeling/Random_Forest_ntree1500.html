Random Forest

n = nrow(data)
K = 10 # 10-fold CV

# 설정하려는 Seed 총 개수
Seed_Num = 10
  
# Seed, parameter 별로 일단 모두 저장
Accuracy_RF <- vector(mode = "list", length = Seed_Num)
FPR_RF <- vector(mode = "list", length = Seed_Num)
FNR_RF <- vector(mode = "list", length = Seed_Num)
F1_RF <- vector(mode = "list", length = Seed_Num)

# Parameter grid 설정
mtry_grid <- c(1, 3, 5, 7, 10, 15)


for(seed in 1:Seed_Num){  # Seed를 다르게 하는 For loop
  print("seed")
  print(seed)
  set.seed(seed)
  
  # Parameter 별로 For loop
  for(mtry in mtry_grid){
    print("mtry")
    print(mtry)
    
      for(CV in 1:K){
        ########################################################################################
        # CV generation. Train : Test를 9:1로 나눔 (이때의 train, test는 validation approach)  #
        ########################################################################################
        CV_index <- split(sample(1:n), rep(1:K, length.out = n))
        train <- data[-CV_index[[CV]], ] # Train set
        train_index <- c(1:n) 
        train_index <- train_index[-CV_index[[CV]]] # Train set의 index (필요할 경우 사용)
        test <- data[CV_index[[CV]], ] # Test set
        test_rowNum <- nrow(test) # Test set row Numbers (필요할 경우 사용))
        
        #################################################################
        # Modeling and Prediction. Update Accuracy, FNR, FPR, F1 score  #
        #################################################################
        # Train index가 필요한 경우
        Model <- randomForest(target ~ ., data = data, subset = train_index, ntree = 1500, mtry = mtry, method = "class")
        
        
        # Prediction (0, 1로 나오는 경우)
        pred <- predict(Model, newdata = test)
        
        # Update Scores
        Accuracy_RF[[seed]] <- c(Accuracy_RF[[seed]], mean(pred == test$target))
        FPR_RF[[seed]] <- c(FPR_RF[[seed]], mean((pred != test$target) & (test$target == 0)) / mean(test$target == 0))
        FNR_RF[[seed]] <- c(FNR_RF[[seed]], mean((pred != test$target) & (test$target == 1)) / mean(test$target == 1))
        F1_RF[[seed]] <- c(F1_RF[[seed]], 2*(mean((pred==test$target)&(test$target==1))/(mean(pred==1))*mean((pred==test$target)&(test$target==1))/(mean(test$target==1))) / (mean((pred==test$target)&(test$target==1))/(mean(pred==1))+mean((pred==test$target)&(test$target==1))/(mean(test$target==1))))
      }
      
    
  }
}

CV meaning process

# Seed, parameter 별로 CV에 대해 평균내어 모두 저장
CVAccuracy_RF <- vector(mode = "list", length = Seed_Num)
CVFPR_RF <- vector(mode = "list", length = Seed_Num)
CVFNR_RF <- vector(mode = "list", length = Seed_Num)
CVF1_RF <- vector(mode = "list", length = Seed_Num)

for(seed in 1:Seed_Num){  # Seed를 다르게 하는 For loop
  # Parameter 별로 For loop
  for(param1 in 1:length(mtry_grid)){
      CVAccuracy_RF[[seed]] <- c(CVAccuracy_RF[[seed]], mean(Accuracy_RF[[seed]][1:K + K * (param1 - 1)]))
      CVFPR_RF[[seed]] <- c(CVFPR_RF[[seed]], mean(FPR_RF[[seed]][1:K + K * (param1 - 1)]))
      CVFNR_RF[[seed]] <- c(CVFNR_RF[[seed]], mean(FNR_RF[[seed]][1:K + K * (param1 - 1)]))
      CVF1_RF[[seed]] <- c(CVF1_RF[[seed]], mean(F1_RF[[seed]][1:K + K * (param1 - 1)]))
  }
}
